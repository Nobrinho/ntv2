# Baseline profile: código pré-compilado (AOT) na instalação.
# Motivo: no Fire TV, cada compilação JIT faz um mprotect no code cache que trava o mapa de memória
# do processo inteiro; com o sistema sob pressão isso chegou a ~1 s e congelou a thread
# ExoPlayer:Playback (underrun de áudio + frame parado). Pré-compilando o caminho da reprodução,
# o JIT praticamente não roda durante o vídeo.

# Player (Media3 + decoder FFmpeg)
HSPLandroidx/media3/**->**(**)**
Landroidx/media3/**;

# Coroutines / Kotlin stdlib (usados pelo coordinator, data sources e UI)
HSPLkotlinx/coroutines/**->**(**)**
Lkotlinx/coroutines/**;
HSPLkotlin/**->**(**)**
Lkotlin/**;

# TDLib (bindings Java usados pelo data source de streaming)
HSPLorg/drinkless/tdlib/**->**(**)**
Lorg/drinkless/tdlib/**;

# Código do app ligado à reprodução
HSPLcom/ntv2/app/core/player/**->**(**)**
Lcom/ntv2/app/core/player/**;
HSPLcom/ntv2/app/feature/playback/**->**(**)**
Lcom/ntv2/app/feature/playback/**;

# Navegação e grades (Fire TV): sem isto a UI do app roda interpretada/JIT até o dexopt em segundo
# plano, e cada compilação JIT na rolagem custava quadros.
HSPLcom/ntv2/app/core/ui/**->**(**)**
Lcom/ntv2/app/core/ui/**;
HSPLcom/ntv2/app/core/navigation/**->**(**)**
Lcom/ntv2/app/core/navigation/**;
HSPLcom/ntv2/app/feature/media/**->**(**)**
Lcom/ntv2/app/feature/media/**;
HSPLcoil/**->**(**)**
Lcoil/**;

# Todo o código do app (Macrobenchmark não roda no Fire TV, API 25 < 28): o Compose e as libs
# trazem os próprios perfis; aqui garantimos que nada do app fique interpretado/JIT.
HSPLcom/ntv2/app/**->**(**)**
Lcom/ntv2/app/**;
